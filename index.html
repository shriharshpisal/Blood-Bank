-- View all active emergency responses
SELECT eq.title, bb.bank_name, er.blood_group,
       er.units_pledged, er.units_dispatched,
       er.status, er.eta_minutes
FROM emergency_responses er
JOIN emergency_requests eq ON er.emergency_id = eq.emergency_id
JOIN blood_banks        bb ON er.bank_id       = bb.bank_id
WHERE eq.status IN ('Active','Coordinating')
ORDER BY eq.severity DESC;

🔗 Compatibility Matrix
-- What can donate to O- recipient?
SELECT donor_group
FROM compatibility_rules
WHERE recipient_group = 'O-'
  AND is_compatible = TRUE;

-- Full matrix summary
SELECT donor_group,
  STRING_AGG(recipient_group, ', ')
    FILTER (WHERE is_compatible)
    AS can_donate_to
FROM compatibility_rules
GROUP BY donor_group;
📊 Bank Performance Report
SELECT bb.bank_name, bb.city,
  COUNT(DISTINCT d.donation_id)
    AS donations,
  COUNT(DISTINCT r.request_id)
    AS requests_handled,
  COUNT(DISTINCT er.response_id)
    AS emergency_responses,
  SUM(bi.units_available) AS stock
FROM blood_banks bb
LEFT JOIN donations d ON bb.bank_id=d.bank_id
LEFT JOIN blood_requests r
       ON bb.bank_id=r.assigned_bank
LEFT JOIN emergency_responses er
       ON bb.bank_id=er.bank_id
LEFT JOIN blood_inventory bi
       ON bb.bank_id=bi.bank_id
GROUP BY bb.bank_id, bb.bank_name, bb.city
ORDER BY emergency_responses DESC;
